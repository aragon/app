const blen = (s) => { let n = 0; for (let i = 0; i < s.length; i++) { const c = s.charCodeAt(i); n += (c === 10 || c === 34 || c === 92) ? 2 : c < 32 ? 6 : c < 128 ? 1 : c < 2048 ? 2 : 3; } return n; };

async function vname(id) {
  if (!(id in vc)) {
    let v = null; try { v = await figma.variables.getVariableByIdAsync(id); } catch (e) {}
    vc[id] = !v ? '?var' : v.variableCollectionId === SEM ? v.name : v.variableCollectionId === PRIM ? 'P:' + v.name : 'O:' + v.name;
  }
  return vc[id];
}
const h2 = (x) => Math.round(x * 255).toString(16).padStart(2, '0');
const hex = (c, a) => '#' + h2(c.r) + h2(c.g) + h2(c.b) + (a !== undefined && a < 1 ? h2(a) : '');
async function sname(sid) {
  if (typeof sid !== 'string' || !sid) return '';
  if (!(sid in sc)) { let s = null; try { s = await figma.getStyleByIdAsync(sid); } catch (e) {} sc[sid] = s ? s.name : '?'; }
  return sc[sid];
}
async function paintToks(n, prop) {
  if (!(prop in n)) return [];
  let ps = n[prop];
  if (ps === figma.mixed) {
    if (n.type === 'TEXT' && prop === 'fills') { ps = []; for (const s of n.getStyledTextSegments(['fills'])) for (const p of s.fills) ps.push(p); }
    else return ['MIXED'];
  }
  if (!Array.isArray(ps)) return [];
  const out = [];
  for (const p of ps) {
    if (p.visible === false) continue;
    if (p.type === 'SOLID') {
      const b = p.boundVariables && p.boundVariables.color;
      let t;
      if (b) t = await vname(b.id);
      else { const st = await sname(prop === 'fills' ? n.fillStyleId : n.strokeStyleId); t = 'RAW:' + hex(p.color) + (st ? '{style ' + st + '}' : ''); }
      if (p.opacity !== undefined && p.opacity < 1) t += '@' + Math.round(p.opacity * 100);
      out.push(t);
    } else if (p.type && p.type.startsWith('GRADIENT')) {
      const ss = [];
      for (const s of p.gradientStops) { const b = s.boundVariables && s.boundVariables.color; ss.push(b ? await vname(b.id) : 'RAW:' + hex(s.color, s.color.a)); }
      out.push('GRAD[' + ss.join('>') + ']');
    } else out.push(p.type);
  }
  return [...new Set(out)];
}
async function effToks(n) {
  if (!('effects' in n) || !Array.isArray(n.effects)) return [];
  const out = [];
  for (const e of n.effects) {
    if (e.visible === false || !e.color) continue;
    const b = e.boundVariables && e.boundVariables.color;
    out.push((e.type === 'INNER_SHADOW' ? 'in:' : '') + (b ? await vname(b.id) : 'RAW:' + hex(e.color, e.color.a)));
  }
  return [...new Set(out)];
}
async function mcName(inst) {
  let mc = null; try { mc = await inst.getMainComponentAsync(); } catch (e) {}
  if (!mc) return '?';
  if (!(mc.id in mcc)) mcc[mc.id] = mc.parent && mc.parent.type === 'COMPONENT_SET' ? mc.parent.name : mc.name;
  return mcc[mc.id];
}
const short = (s) => (s.length > 22 ? s.slice(0, 21) + '~' : s);
// walk one variant; items is a Set of strings
async function walk(n, path, items, isRoot) {
  if (!isRoot && n.visible === false) return;
  const lp = isRoot ? '@' : path.slice(-2).map(short).join('/');
  if (!isRoot && n.type === 'INSTANCE') {
    items.add(lp + '=>' + (await mcName(n)));
    // colour overrides set on the instance (own node or its sublayers)
    let ov = []; try { ov = n.overrides || []; } catch (e) {}
    for (const o of ov) {
      const f = o.overriddenFields.filter((x) => x === 'fills' || x === 'strokes' || x === 'effects');
      if (!f.length) continue;
      let t = null; try { t = await figma.getNodeByIdAsync(o.id); } catch (e) {}
      if (!t || t.visible === false) continue;
      const sub = t.id === n.id ? '' : '>' + short(t.name);
      const tag = t.type === 'TEXT' ? 'T' : null;
      if (f.includes('fills')) for (const k of await paintToks(t, 'fills')) items.add(lp + sub + ' *' + (tag || 'F') + ' ' + k);
      if (f.includes('strokes')) for (const k of await paintToks(t, 'strokes')) items.add(lp + sub + ' *S ' + k);
      if (f.includes('effects')) for (const k of await effToks(t)) items.add(lp + sub + ' *E ' + k);
    }
    return;
  }
  const ft = n.type === 'TEXT' ? 'T' : 'F';
  for (const k of await paintToks(n, 'fills')) items.add(lp + ' ' + ft + ' ' + k);
  for (const k of await paintToks(n, 'strokes')) items.add(lp + ' S ' + k);
  for (const k of await effToks(n)) items.add(lp + ' E ' + k);
  if ('children' in n) for (const c of n.children) await walk(c, path.concat([c.name]), items, false);
}
function vprops(v) {
  try { if (v.variantProperties) return v.variantProperties; } catch (e) {}
  const o = {}; for (const kv of v.name.split(',')) { const i = kv.indexOf('='); if (i > 0) o[kv.slice(0, i).trim()] = kv.slice(i + 1).trim(); } return o;
}
function describe(vps, allVals) {
  const keys = Object.keys(allVals);
  const sets = {}; for (const k of keys) sets[k] = new Set(vps.map((x) => x[k]));
  let prod = 1; for (const k of keys) prod *= sets[k].size;
  const fixed = keys.filter((k) => sets[k].size < allVals[k].size);
  if (prod === vps.length) return fixed.length ? fixed.map((k) => k + '=' + [...sets[k]].join('/')).join('; ') : '(all)';
  const one = keys.filter((k) => sets[k].size === 1);
  const vary = keys.filter((k) => sets[k].size > 1);
  const head = one.filter((k) => allVals[k].size > 1).map((k) => k + '=' + [...sets[k]][0]).join('; ');
  const combos = vps.slice(0, 10).map((x) => vary.map((k) => x[k]).join(',')).join(' | ');
  return (head ? head + ' :: ' : '') + '[' + vary.join(',') + '] ' + combos + (vps.length > 10 ? ' | +' + (vps.length - 10) + ' more' : '');
}
